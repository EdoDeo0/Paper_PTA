\begin{frame}{Research Question \& Data}

  \textbf{Goal:} does the environmental content of China's PTAs shift
  exports towards green goods and away from dirty goods?

  \vspace{1.2em}
  \begin{itemize}\setlength\itemsep{10pt}
    \item Chinese customs transactions, 2000--2015 \hfill \small(45.8M obs.)
    \item 14 PTAs signed by China \hfill \small(23 treated destinations)
    \item EP content: World Bank DTA \& TREND databases
    \item Green/dirty goods: OECD CLEG list \& pollution-intensity mapping
    \item Robustness samples: HS4-restricted comparison, CEM-matched destinations
  \end{itemize}

\end{frame}

% ======================================================================
\begin{frame}{Empirical Analysis}

  \begin{center}
  \[
  \ln x_{fpdt} = \beta_1\, EP_{dt}\!\times\!g_p + \beta_2\, EP_{dt}\!\times\!b_p
  + \gamma_1\, TD_{dt}\!\times\!g_p + \gamma_2\, TD_{dt}\!\times\!b_p
  + \text{FE} + \varepsilon_{fpdt}
  \]
  \end{center}

  \small
  \begin{center}
  firm $\times$ product $\times$ destination $\times$ year, with
  firm--destination--year fixed effects
  \end{center}
  \normalsize

  \vspace{1.2em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{\small Three levels of inference}
      \begin{itemize}\setlength\itemsep{4pt}
        \item Asymptotic
        \item Wild cluster bootstrap
        \item Permutation
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{\small Robustness of measurement}
      \begin{itemize}\setlength\itemsep{4pt}
        \item WB \emph{and} TREND indices
        \item Continuous CO$_2$ intensity
      \end{itemize}
    \end{column}
  \end{columns}

\end{frame}
